.vendor = "unknown"

parsed, err = parse_cef(.message)

if err == null {
    . |= parsed

    if exists(.deviceVendor) {

        vendor = downcase(string!(.deviceVendor))

        if vendor == "palo alto networks" {
            .vendor = "paloalto"
        } else if vendor == "imperva" {
            .vendor = "waf"
        } else if vendor == "f5" {
            .vendor = "waf"
        } else if vendor == "fortinet" && exists(.deviceProduct) && contains(downcase(string!(.deviceProduct)), "fortiweb") {
            .vendor = "waf"
        }
    }
}
